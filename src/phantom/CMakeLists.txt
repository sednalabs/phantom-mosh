cmake_minimum_required(VERSION 3.16)
project(phantom_record LANGUAGES CXX)
find_package(OpenSSL 1.1.1 REQUIRED COMPONENTS Crypto)
option(PHANTOM_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_library(phantom_record STATIC record.cc startup.cc)
target_compile_features(phantom_record PUBLIC cxx_std_17)
target_include_directories(phantom_record PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(phantom_record PUBLIC OpenSSL::Crypto)
if(MSVC)
  target_compile_options(phantom_record PRIVATE /W4 /WX)
else()
  target_compile_options(phantom_record PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion)
  if(PHANTOM_SANITIZERS)
    target_compile_options(phantom_record PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(phantom_record PUBLIC -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
add_executable(phantom_record_test ../../tests/phantom/record_test.cc)
target_link_libraries(phantom_record_test PRIVATE phantom_record)
add_test(NAME phantom-record COMMAND phantom_record_test)
set_tests_properties(phantom-record PROPERTIES TIMEOUT 60)
if(UNIX)
  target_sources(phantom_record PRIVATE startup_posix.cc)
  add_executable(phantom_udp_test ../../tests/phantom/udp_test.cc)
  target_link_libraries(phantom_udp_test PRIVATE phantom_record)
  add_test(NAME phantom-udp COMMAND phantom_udp_test)
  set_tests_properties(phantom-udp PROPERTIES TIMEOUT 15)
endif()
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
add_test(NAME phantom-independent-reference
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/phantom/reference.py $<TARGET_FILE:phantom_record_test>)
set_tests_properties(phantom-independent-reference PROPERTIES TIMEOUT 15)
add_test(NAME phantom-evaluation-guards
  COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/../../evaluation -p test_observer.py)
set_tests_properties(phantom-evaluation-guards PROPERTIES TIMEOUT 15)

add_executable(phantom_startup_test ../../tests/phantom/startup_test.cc)
target_link_libraries(phantom_startup_test PRIVATE phantom_record)
add_test(NAME phantom-startup COMMAND phantom_startup_test)
set_tests_properties(phantom-startup PROPERTIES TIMEOUT 15)
if(UNIX)
  add_executable(phantom_startup_process ../../tests/phantom/startup_process.cc)
  target_link_libraries(phantom_startup_process PRIVATE phantom_record)
  add_test(NAME phantom-startup-process
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/phantom/startup_process_test.py $<TARGET_FILE:phantom_startup_process>)
  set_tests_properties(phantom-startup-process PROPERTIES TIMEOUT 30)
endif()

set(PHANTOM_SSH_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(PHANTOM_SSH_DEFAULT ON)
endif()
option(PHANTOM_BUILD_SSH_STARTUP "Build the Linux SSH startup supervisor" ${PHANTOM_SSH_DEFAULT})
option(PHANTOM_REQUIRE_SSH_TESTS "Require real OpenSSH integration test prerequisites" OFF)
if(PHANTOM_REQUIRE_SSH_TESTS AND NOT PHANTOM_BUILD_SSH_STARTUP)
  message(FATAL_ERROR "PHANTOM_REQUIRE_SSH_TESTS requires PHANTOM_BUILD_SSH_STARTUP")
endif()
if(PHANTOM_BUILD_SSH_STARTUP)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The SSH startup supervisor currently supports Linux only")
  endif()
  include(CheckCXXSymbolExists)
  set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
  check_cxx_symbol_exists(posix_spawn_file_actions_addclosefrom_np "spawn.h" PHANTOM_HAVE_SPAWN_CLOSEFROM)
  unset(CMAKE_REQUIRED_DEFINITIONS)
  if(NOT PHANTOM_HAVE_SPAWN_CLOSEFROM)
    message(FATAL_ERROR "SSH startup requires glibc 2.34+ spawn closefrom support; disable PHANTOM_BUILD_SSH_STARTUP otherwise")
  endif()
  add_library(phantom_ssh_startup STATIC ssh_startup.cc)
  target_link_libraries(phantom_ssh_startup PUBLIC phantom_record)
  target_compile_definitions(phantom_ssh_startup PRIVATE _GNU_SOURCE)
  target_compile_options(phantom_ssh_startup PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion)
  add_executable(phantom_ssh_driver ../../tests/phantom/ssh_driver.cc)
  target_link_libraries(phantom_ssh_driver PRIVATE phantom_ssh_startup)
  add_test(NAME phantom-ssh-process
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/phantom/ssh_process_test.py $<TARGET_FILE:phantom_ssh_driver>)
  set_tests_properties(phantom-ssh-process PROPERTIES TIMEOUT 45)
  set(PHANTOM_SSH_TEST_ARGUMENTS)
  if(PHANTOM_REQUIRE_SSH_TESTS)
    list(APPEND PHANTOM_SSH_TEST_ARGUMENTS --required)
  endif()
  add_test(NAME phantom-ssh-integration
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/phantom/ssh_integration_test.py
      $<TARGET_FILE:phantom_ssh_driver> ${PHANTOM_SSH_TEST_ARGUMENTS})
  set_tests_properties(phantom-ssh-integration PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
endif()
